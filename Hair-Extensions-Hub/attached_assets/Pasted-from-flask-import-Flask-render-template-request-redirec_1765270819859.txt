os.makedirs("responses", exist_ok=True)


@app.route("/", methods=["GET"])

    # TODO: later, verify a token or payment flag here if you only want paid clients to access

        "client_name": request.form.get("client_name", "").strip(),
        "email": request.form.get("email", "").strip(),
        "country": request.form.get("country", "").strip(),
        "state_region": request.form.get("state_region", "").strip(),
        "city": request.form.get("city", "").strip(),
        "suburb": request.form.get("suburb", "").strip(),
        "nearest_hub": request.form.get("nearest_hub", "").strip(),
        "travel_radius": request.form.get("travel_radius", "").strip(),
        "search_channels": request.form.getlist("search_channels"),
        "main_goal": request.form.get("main_goal", "").strip(),
        "dream_hair": request.form.get("dream_hair", "").strip(),
        "hair_length": request.form.get("hair_length", "").strip(),
        "hair_thickness": request.form.get("hair_thickness", "").strip(),
        "hair_texture": request.form.get("hair_texture", "").strip(),
        "hair_colour": request.form.get("hair_colour", "").strip(),
        "changing_colour": request.form.get("changing_colour", "").strip(),
        "wash_frequency": request.form.get("wash_frequency", "").strip(),
        "heat_styling": request.form.get("heat_styling", "").strip(),
        "activity_level": request.form.get("activity_level", "").strip(),
        "maintenance_level": request.form.get("maintenance_level", "").strip(),
        "install_budget": request.form.get("install_budget", "").strip(),
        "maintenance_budget": request.form.get("maintenance_budget", "").strip(),
        "had_extensions_before": request.form.get("had_extensions_before", "").strip(),
        "previous_methods": request.form.getlist("previous_methods"),
        "previous_issues": request.form.getlist("previous_issues"),

        "stylists_considering": request.form.get("stylists_considering", "").strip(),

        json.dump(data, f, ensure_ascii=False, indent=2)

    # TODO: Here is where you'll:

    return redirect(url_for("thank_you", client_name=data["client_name"]))
